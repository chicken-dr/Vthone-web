Language Features

Clean syntax, Python interoperability, and modern language features — all transpiling to standard CPython

Variables & Constants

Mutable bindings with let, compile-time protected constants with const.

let - Mutable Variables

Vthon
let x = 10
let name = "Vthon"
let enabled = true
let nothing = null
let values = [1, 2, 3]
let user = {"name": "Ada"}

# Reassignment allowed
x = 20
values.append(4)

Type Annotations (Optional)

let x: int = 10
let name: str = "Vthon"
let items: list = [1, 2, 3]
let data: dict = {"key": "value"}

const - Compile-Time Constants

const PI = 3.14159
const MAX_SIZE = 100
const DEFAULT_NAME = "unknown"
const FLAGS = [true, false, true]

# Compile-time error on reassignment:
# PI = 3.14  # ERROR: cannot assign to constant 'PI'

Note: Constants are compile-time protection only. They are not a security boundary against mutation through aliases or Python reflection.

Vthon Literals

VthonPython
trueTrue
falseFalse
nullNone

Control Flow

Familiar conditional and loop constructs with indentation-based blocks.

If / Elif / Else

let x = 10

if x > 5:
    print("large")
elif x == 5:
    print("equal")
else:
    print("small")

# Ternary-like (expression)
let result = "positive" if x > 0 else "negative"

While Loops

let n = 5
while n > 0:
    print(n)
    n -= 1

# With break/continue
while true:
    let input = get_input()
    if input == "quit":
        break
    if input == "":
        continue
    process(input)

For Loops

# Range iteration
for i in range(5):
    print(i)  # 0, 1, 2, 3, 4

# Collection iteration
for name in ["Ada", "Alan", "Grace"]:
    print(name)

# Dictionary iteration
for key, value in {"a": 1, "b": 2}.items():
    print(key, value)

# With enumerate
for i, item in enumerate(items):
    print(i, item)

Loop Control

for i in range(10):
    if i == 3:
        continue  # Skip 3
    if i == 7:
        break     # Stop at 7
    print(i)

# pass as placeholder
if condition:
    pass  # TODO: implement

Functions

First-class functions with parameters, return values, and optional type annotations.

Basic Functions

fn add(a, b):
    return a + b

fn greet(name: str) -> str:
    return "Hello, " + name

fn noop():
    pass

print(add(2, 3))     # 5
print(greet("Ada"))  # Hello, Ada

Parameters

# Positional & keyword
fn example(a, b=10):
    return a + b

example(1)       # 11
example(1, 2)    # 3
example(a=5)     # 15

# *args, **kwargs
fn variadic(*args, **kwargs):
    print(args, kwargs)

variadic(1, 2, x=3, y=4)

Return Values

fn add(a, b):
    return a + b

fn early_return(x):
    if x < 0:
        return  # Returns null
    return x * 2

fn implicit_return():
    42  # Last expression (not auto-returned)

# Multiple returns
fn divmod(a, b):
    return a // b, a % b

let q, r = divmod(10, 3)

Type Annotations

fn add(a: int, b: int) -> int:
    return a + b

fn greet(name: str) -> str:
    return "Hello, " + name

fn process(items: list) -> dict:
    return {"count": len(items)}

Classes

Object-oriented programming with implicit self, inheritance, and Python-compatible initialization.

Basic Classes

class Person:
    fn init(name):
        self.name = name

    fn greet():
        return "Hello, " + self.name

    fn set_age(age):
        self.age = age

let person = Person("Ada")
person.set_age(36)
print(person.greet())  # Hello, Ada

Inheritance

class Employee(Person):
    fn init(name, role):
        super().init(name)
        self.role = role

    fn greet():
        return super().greet() + " (Employee)"

let emp = Employee("Alan", "Engineer")
print(emp.greet())  # Hello, Alan (Employee)

Implicit Self Rules

class Example:
    # Implicit self added (no self/cls as first param)
    fn method():
        print(self)

    # Explicit self - no implicit added
    fn explicit(self):
        print(self)

    # Class method (cls as first param)
    fn classmethod(cls):
        print(cls)

    # Static-like (no implicit, but not decorated)
    fn static():
        print("static")

Initialization

fn init(...) inside a class transpiles to Python's __init__:

# Vthon
class Point:
    fn init(x, y):
        self.x = x
        self.y = y

# Generated Python
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

Exception Handling

Full try/catch/finally support with Python-compatible semantics.

Try / Catch / Finally

try:
    raise ValueError("bad value")
catch ValueError as error:
    print("Caught:", error)
catch TypeError:
    print("Type error")
catch:
    print("Other exception")
finally:
    print("Cleanup always runs")

Rules

  • At least one catch or finally required
  • catch without type catches all exceptions
  • as name binds exception (optional)
  • Multiple catch clauses allowed

Raise

# Raise exception
raise ValueError("invalid input")

# Raise with existing exception
try:
    risky()
catch as error:
    raise  # Re-raise

# Raise any expression
raise "string exception"  # Works (Python allows)

Transpilation

# Vthon
try:
    raise ValueError("bad")
catch ValueError as error:
    print(error)
finally:
    print("done")

# Generated Python
try:
    raise ValueError("bad")
except ValueError as error:
    print(error)
finally:
    print("done")

Python Interoperability

Explicit py import for seamless Python ecosystem access.

Import Syntax

# Explicit Python import
py import requests
py import numpy as np
py from pathlib import Path
py from json import dumps, loads

# Plain imports also work (treated as py import)
import math
from datetime import datetime

Usage

py import requests
py from json import dumps

let response = requests.get("https://api.github.com")
print(response.status_code)

let data = {"user": "Ada", "active": true}
let json_str = dumps(data, sort_keys=true)
print(json_str)

How It Works

  • py import statements pass through to generated Python unchanged
  • Imported Python objects are available directly in Vthon scope
  • Vthon values use Python runtime representations (no wrapping)
  • Call Python functions, access attributes, pass Vthon values to Python

Popular Packages

# Web requests
py import requests

# Data science
py import numpy as np
py import pandas as pd

# File system
py from pathlib import Path

# Async
py import asyncio

# All work natively!

Developer Tooling

Built-in CLI, REPL, and diagnostic tools.

CLI Commands

vthon file.vth        # Run file
vthon run file.vth    # Explicit run
vthon repl            # Start REPL
vthon --emit-python file.vth  # Show generated Python
vthon --version       # Show version

REPL Features

  • Multi-line block support (blank line executes)
  • Single-line immediate execution
  • Persistent namespace
  • Ctrl-D / Ctrl-C to exit
>> let x = 10
>>> print(x)
10
>>> fn double(n):
...     return n * 2
...
>>> print(double(21))
42

Diagnostics

Source-located errors for all phases:

# Syntax error
hello.vth:2:1: error: indentation must be multiple of 4
  if true:
  print(1)
  ^

# Compile error (const reassignment)
hello.vth:3:5: error: cannot assign to constant 'PI'
  const PI = 3.14
  PI = 3.14
    ^

# Runtime error (Python exception)
hello.vth:5:1: runtime error: ZeroDivisionError: division by zero
  print(1 / 0)
  ^

Programmatic API

import vthon.runtime as vthon

# Parse to AST
ast = vthon.parse(source)

# Transpile only
python = vthon.transpile(source)

# Execute with custom namespace
ns = vthon.execute(source, namespace={"x": 10})

# Run file
ns = vthon.run_file("program.vth")
