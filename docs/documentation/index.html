Example Programs

Learn by example - all included in the repository

hello.vth

Basic output and variables

let name = "Vthon"
print("Hello from " + name + "!")

variables.vth

let/const, types, collections

let language = "Vthon"
const VERSION = "0.2.0"
let features = ["readable", "indented"]
print(language + " " + VERSION)

functions.vth

Function definitions

fn add(a, b):
    return a + b

print(add(20, 22))

classes.vth

Classes with implicit self

class Person:
    fn init(name):
        self.name = name

let person = Person("Ada")
print(person.greet())

python_import.vth

Python interoperability

py import math
py from json import dumps

let result = {"sqrt": math.sqrt(81)}
print(dumps(result))

control_flow.vth

if/elif/else statements

let x = 10
if x > 5:
    print("large")
elif x == 5:
    print("equal")
else:
    print("small")
